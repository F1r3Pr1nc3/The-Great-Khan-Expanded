#########################################
# Khan Planet Development — Scripted Triggers
# Used by khan_develop_planet to check zone building capacity.
# Each zone has 3 building slots. A planet can have 1 or 2 zones of the same type.
# These triggers count all buildings that could occupy a zone's slots (planet-wide)
# and check against the 3-slot (1 zone) or 6-slot (2+ zones) capacity.
#########################################

# True if planet has 2+ research-type zones (each adds 3 building slots)
khan_has_multiple_research_zones = {
	calc_true_if = {
		amount >= 2
		has_zone = { zone = zone_research }
		has_zone = { zone = zone_research_unity }
		has_zone = { zone = zone_research_physics }
		has_zone = { zone = zone_research_society }
		has_zone = { zone = zone_research_engineering }
	}
}

# Counts all research-capable buildings (planet-wide) and checks >= 3
# Includes: research_lab_1/2/3, physics_lab_1/2/3, biolab_1/2/3, engineering_facility_1/2/3
khan_research_count_ge_3 = {
	calc_true_if = {
		amount >= 3
		# Focused research buildings (planet_limit=1, at most 1 of any tier per type)
		num_buildings = { type = building_physics_lab_1 value >= 1 }
		num_buildings = { type = building_physics_lab_2 value >= 1 }
		num_buildings = { type = building_physics_lab_3 value >= 1 }
		num_buildings = { type = building_biolab_1 value >= 1 }
		num_buildings = { type = building_biolab_2 value >= 1 }
		num_buildings = { type = building_biolab_3 value >= 1 }
		num_buildings = { type = building_engineering_facility_1 value >= 1 }
		num_buildings = { type = building_engineering_facility_2 value >= 1 }
		num_buildings = { type = building_engineering_facility_3 value >= 1 }
		# Generic research labs (no planet_limit, up to 3 per tier needed for this check)
		num_buildings = { type = building_research_lab_1 value >= 1 }
		num_buildings = { type = building_research_lab_1 value >= 2 }
		num_buildings = { type = building_research_lab_1 value >= 3 }
		num_buildings = { type = building_research_lab_2 value >= 1 }
		num_buildings = { type = building_research_lab_2 value >= 2 }
		num_buildings = { type = building_research_lab_2 value >= 3 }
		num_buildings = { type = building_research_lab_3 value >= 1 }
		num_buildings = { type = building_research_lab_3 value >= 2 }
		num_buildings = { type = building_research_lab_3 value >= 3 }
	}
}

# Same count but checks >= 6 (for planets with 2+ research zones)
khan_research_count_ge_6 = {
	calc_true_if = {
		amount >= 6
		# Focused research buildings
		num_buildings = { type = building_physics_lab_1 value >= 1 }
		num_buildings = { type = building_physics_lab_2 value >= 1 }
		num_buildings = { type = building_physics_lab_3 value >= 1 }
		num_buildings = { type = building_biolab_1 value >= 1 }
		num_buildings = { type = building_biolab_2 value >= 1 }
		num_buildings = { type = building_biolab_3 value >= 1 }
		num_buildings = { type = building_engineering_facility_1 value >= 1 }
		num_buildings = { type = building_engineering_facility_2 value >= 1 }
		num_buildings = { type = building_engineering_facility_3 value >= 1 }
		# Generic research labs (up to 6 per tier needed for this check)
		num_buildings = { type = building_research_lab_1 value >= 1 }
		num_buildings = { type = building_research_lab_1 value >= 2 }
		num_buildings = { type = building_research_lab_1 value >= 3 }
		num_buildings = { type = building_research_lab_1 value >= 4 }
		num_buildings = { type = building_research_lab_1 value >= 5 }
		num_buildings = { type = building_research_lab_1 value >= 6 }
		num_buildings = { type = building_research_lab_2 value >= 1 }
		num_buildings = { type = building_research_lab_2 value >= 2 }
		num_buildings = { type = building_research_lab_2 value >= 3 }
		num_buildings = { type = building_research_lab_2 value >= 4 }
		num_buildings = { type = building_research_lab_2 value >= 5 }
		num_buildings = { type = building_research_lab_2 value >= 6 }
		num_buildings = { type = building_research_lab_3 value >= 1 }
		num_buildings = { type = building_research_lab_3 value >= 2 }
		num_buildings = { type = building_research_lab_3 value >= 3 }
		num_buildings = { type = building_research_lab_3 value >= 4 }
		num_buildings = { type = building_research_lab_3 value >= 5 }
		num_buildings = { type = building_research_lab_3 value >= 6 }
	}
}

# True if planet has room for more research buildings in research zones
# 1 zone = limit 3, 2+ zones = limit 6
khan_research_slots_available = {
	OR = {
		AND = {
			khan_has_multiple_research_zones = yes
			NOT = { khan_research_count_ge_6 = yes }
		}
		AND = {
			NOT = { khan_has_multiple_research_zones = yes }
			NOT = { khan_research_count_ge_3 = yes }
		}
	}
}
